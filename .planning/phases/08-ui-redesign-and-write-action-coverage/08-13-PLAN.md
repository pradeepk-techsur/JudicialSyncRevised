Renders every `objections[]` row (UNRESOLVED threads, zero/one/several per F10 §Process step 4) with objecting party, grounds, elapsed time since `raisedAt`, and its OWN `RecordRulingForm` passing THAT row's `objectionId` — never ambiguous about which thread a ruling applies to, even when several threads exist concurrently:

```tsx
'use client';
import { useState } from 'react';
import { Tile } from '@carbon/react';
import type { ObjectionCurrentState } from '@prisma/client';
import { RecordRulingForm } from '@/components/actions/RecordRulingForm';
import styles from './ObjectionCard.module.scss';

function elapsed(since: Date): string {
  const mins = Math.round((Date.now() - since.getTime()) / 60000);
  if (mins < 60) return `${mins}m ago`;
  const hours = Math.round(mins / 60);
  return `${hours}h ago`;
}

export function ObjectionCard({ objections }: { objections: ObjectionCurrentState[] }) {
  const [openId, setOpenId] = useState<string | null>(null);

  return (
    <Tile data-testid="exhibit-objection-card" aria-label="Objection">
      <h2 className={styles.heading}>Objection</h2>
      {objections.length === 0 ? (
        <p className={styles.empty} data-testid="objection-card-empty">No open objections</p>
      ) : (
        objections.map((obj) => (
          <div key={obj.objectionId} className={styles.thread} data-testid="objection-thread" data-objection-id={obj.objectionId}>
            <p className={styles.grounds}>{obj.objectingParty}: {obj.grounds}</p>
            <p className={styles.caption}>
              Raised {elapsed(new Date(obj.raisedAt))}. Only the judge can rule.
            </p>
            {openId !== obj.objectionId ? (
              <button type="button" data-testid="objection-record-ruling-trigger" onClick={() => setOpenId(obj.objectionId)}>
                Record ruling
              </button>
            ) : (
              <RecordRulingForm objectionId={obj.objectionId} onDone={() => setOpenId(null)} />
            )}
          </div>
        ))
      )}
    </Tile>
  );
}
```

```tsx
export function JuryPackageChecklistCard({
  checklist,
  exhibitId,
}: { checklist: ExhibitHistoryResponse['juryPackageChecklist']; exhibitId: string }) {
  const items = [
    { met: checklist.admitted, label: 'Admitted' },
    { met: checklist.objectionsResolved, label: 'No unresolved objections' },
    { met: checklist.custodianOnRecord, label: 'Custodian on record' },
    { met: checklist.classificationTrial, label: 'Not sealed / ex parte' },
  ];
  const eligibilityTone = checklist.eligibility === 'INCLUDED' ? 'medium' : checklist.eligibility === 'BLOCKED' ? 'high' : 'pending';
  return (
    <Tile data-testid="exhibit-jury-checklist-card" aria-label="Jury package">
      <h2 className={styles.heading}>Jury Package</h2>
      <SeverityPill tone={eligibilityTone} label={checklist.eligibility === 'NOT_ELIGIBLE' ? 'Not eligible' : checklist.eligibility === 'INCLUDED' ? 'Included' : 'Blocked'} />
      <ul className={styles.checklist}>
        {items.map((item) => (
          <li key={item.label} data-testid="jury-checklist-item" data-met={item.met}>
            {item.met ? '✓' : '✗'} {item.label}
          </li>
        ))}
      </ul>
      <Link href="/jury-package" data-testid="jury-checklist-open-link">Open jury package →</Link>
    </Tile>
  );
}
```
