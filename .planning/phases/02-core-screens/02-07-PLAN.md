```typescript
'use client';

import { useQuery } from '@tanstack/react-query';
import { apiFetch } from '@/lib/apiClient';
import { useRoleStore } from '@/stores/roleStore';
import type { ExhibitHistoryResponse } from '@/services/history';

export class NotFoundError extends Error {}

export function useExhibitHistory(exhibitId: string) {
  const role = useRoleStore((s) => s.role); // in the query key: switching role must re-fetch (sealed exhibit should flip to not-found immediately, not wait for the next poll)

  return useQuery<ExhibitHistoryResponse>({
    queryKey: ['exhibit-history', exhibitId, role],
    queryFn: async () => {
      const res = await apiFetch(`/api/exhibits/${exhibitId}/history`);
      if (res.status === 404) {
        throw new NotFoundError('Exhibit not found');
      }
      if (!res.ok) {
        throw new Error(`Failed to load exhibit history: ${res.status}`);
      }
      return res.json();
    },
    retry: (failureCount, error) => !(error instanceof NotFoundError) && failureCount < 2,
    // Live-sync polling (Y0-patterns.md) — a custody transfer or new timeline
    // entry recorded elsewhere must appear without manual refresh.
    refetchInterval: 4_000,
  });
}
```
`retry` explicitly disables react-query's default retry behavior for the `NotFoundError` case — a genuinely 404 response must render the not-found state immediately, not after 2 silent retries that would make a sealed-exhibit probe take visibly longer than a real 404 (a timing side-channel the anti-enumeration guarantee must not reopen).

```typescript
  );
```

```typescript
'use client';

import { use } from 'react';
import Link from 'next/link';
import { useExhibitHistory, NotFoundError } from '@/hooks/useExhibitHistory';
import { ExhibitHeader } from '@/components/exhibit/ExhibitHeader';
import { Timeline } from '@/components/exhibit/Timeline';
import { ExhibitNotFound } from '@/components/exhibit/ExhibitNotFound';

export default function ExhibitDetailPage({ params }: { params: Promise<{ id: string }> }) {
  const { id } = use(params);
  const { data, isLoading, error } = useExhibitHistory(id);

  if (error instanceof NotFoundError) {
    return <ExhibitNotFound />;
  }
  if (isLoading) {
    return <p className="text-sm text-gray-500">Loading exhibit history…</p>;
  }
  if (error || !data) {
    return <p className="text-sm text-red-600">Unable to load exhibit history — please retry.</p>;
  }

  return (
    <div>
      <Link href="/case" className="mb-4 inline-block text-sm underline">
        ← Back to Case Workspace
      </Link>
      <ExhibitHeader data={data} />
      <h2 className="mb-2 text-lg font-medium">History</h2>
      <Timeline entries={data.timeline} />
    </div>
  );
}
```

```typescript
import { test, expect } from '@playwright/test';

async function getExhibitId(request: any, label: string, role = 'JUDGE'): Promise<string> {
  const caseRes = await request.get('/api/case');
  const { case: kase } = await caseRes.json();
  const listRes = await request.get(`/api/cases/${kase.id}/exhibits`, {
    headers: { 'X-User-Role': role },
  });
  const rows = await listRes.json();
  const row = rows.find((r: { exhibitLabel: string }) => r.exhibitLabel === label);
  if (!row) throw new Error(`Seed fixture ${label} not found under role ${role}`);
  return row.exhibitId;
}

test.describe('Exhibit Detail View', () => {
  test('header shows status/custodian/party/witness above the fold; full timeline renders in order', async ({ page, request }) => {
    const id = await getExhibitId(request, 'P-1');
    await page.goto(`/exhibit/${id}`);
    await expect(page.getByText('Objected')).toBeVisible();
    await expect(page.getByText(/PROSECUTION/)).toBeVisible();
    // P-1's seeded timeline: MARKED, OFFERED, OBJECTION_RAISED, STATUS_CHANGE->OBJECTED = 4 entries, no ruling.
    const entries = page.locator('[aria-label="Exhibit history timeline"] li');
    await expect(entries).toHaveCount(4);
    await expect(entries.first()).toContainText('Status changed from');
  });

  test('a genuinely nonexistent id and a sealed exhibit under an unauthorized role render the identical not-found page', async ({ page, request }) => {
    const sealedId = await getExhibitId(request, 'S-1', 'JUDGE');

    await page.goto('/exhibit/00000000-0000-0000-0000-000000000000');
    const missingHtml = await page.locator('body').innerText();

    // Switch to ATTORNEY (no sealed visibility) before visiting the sealed id.
    await page.goto('/case');
    const roleSelect = page.getByLabel('Switch active role');
    const attorneyOption = roleSelect.locator('option', { hasText: 'ATTORNEY' });
    await roleSelect.selectOption(await attorneyOption.getAttribute('value') as string);
    await page.goto(`/exhibit/${sealedId}`);
    const sealedHtml = await page.locator('body').innerText();

    expect(sealedHtml).toBe(missingHtml);
    await expect(page.getByText('Exhibit not found')).toBeVisible();
  });

  test('back link returns to /case', async ({ page, request }) => {
    const id = await getExhibitId(request, 'P-1');
    await page.goto(`/exhibit/${id}`);
    await page.getByRole('link', { name: /Back to Case Workspace/ }).click();
    await expect(page).toHaveURL(/\/case$/);
  });

  test('cross-screen parity: status shown here matches the Case Workspace row for the same exhibit', async ({ page, request }) => {
    const id = await getExhibitId(request, 'P-3');
    await page.goto('/case');
    const row = page.locator('[data-testid="exhibit-row"][data-exhibit-label="P-3"]');
    const workspaceStatusText = await row.getByText(/Admitted|Objected|Offered|Marked|Excluded|Withdrawn/).textContent();

    await page.goto(`/exhibit/${id}`);
    await expect(page.getByText(workspaceStatusText!.trim())).toBeVisible();
  });
});
```
