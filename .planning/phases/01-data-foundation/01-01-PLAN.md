  <action>
Write `prisma/schema.prisma` with the EXACT models from FRD `Y0-schema.md` §Core Entities, §Event Ledger, §Current-State Projections — scoped to Phase 1's requirements (F0, F1, F2, F3) only. Two necessary completions beyond Y0-schema.md's excerpt (required for Prisma schema validity, not a specification change — every field name, type, and constraint below is copied verbatim from Y0-schema.md and TechArch §3's canonical SQL DDL):
1. Add back-relation fields on `User` for `ExhibitEvent.actorUserId` and `CustodyCurrentState.currentCustodianUserId` (Prisma requires both sides of every relation declared; Y0-schema.md's `User` excerpt omits these for brevity).
2. Add `@@map`/`@map` directives throughout so the actual Postgres table/column names are `snake_case` exactly matching TechArch §3.2–3.4's canonical SQL DDL (`exhibit_events`, `exhibit_label`, `sequence_no`, etc.), while Prisma-side model/field names stay `PascalCase`/`camelCase` per Y0-schema.md — this is precisely the mapping TechArch §3.8 "ORM mapping" specifies, not an abstraction.

```prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

enum Role {
  JUDGE
  CHAMBERS_STAFF
  DEPUTY
  CLERK
  ATTORNEY
  ADMIN

  @@map("role_type")
}

enum OfferingParty {
  PLAINTIFF
  PROSECUTION
  DEFENSE

  @@map("offering_party")
}

enum ExhibitStatus {
  MARKED
  OFFERED
  OBJECTED
  ADMITTED
  EXCLUDED
  WITHDRAWN

  @@map("exhibit_status")
}

enum ObjectionStatus {
  UNRESOLVED
  SUSTAINED
  OVERRULED

  @@map("objection_status")
}

enum EventType {
  STATUS_CHANGE
  OBJECTION_RAISED
  RULING_RECORDED
  CUSTODY_TRANSFER
  DISCREPANCY_ACKNOWLEDGED

  @@map("event_type")
}

model Case {
  id         String   @id @default(uuid())
  caseNumber String   @unique @map("case_number")
  title      String
  court      String
  createdAt  DateTime @default(now()) @map("created_at")

  users    User[]
  exhibits Exhibit[]

  @@map("cases")
}

model User {
  id        String   @id @default(uuid())
  caseId    String   @map("case_id")
  name      String
  role      Role
  isActive  Boolean  @default(true) @map("is_active")
  createdAt DateTime @default(now()) @map("created_at")

  case               Case                  @relation(fields: [caseId], references: [id])
  actedEvents        ExhibitEvent[]        @relation("EventActor")
  custodyAssignments CustodyCurrentState[] @relation("CurrentCustodian")

  @@index([caseId, role])
  @@map("users")
}

model Exhibit {
  id                String        @id @default(uuid())
  caseId            String        @map("case_id")
  exhibitLabel      String        @map("exhibit_label")
  description       String
  source            String?
  offeringParty     OfferingParty @map("offering_party")
  associatedWitness String?       @map("associated_witness")
  isSealed          Boolean       @default(false) @map("is_sealed")
  createdAt         DateTime      @default(now()) @map("created_at")

  case         Case                    @relation(fields: [caseId], references: [id])
  events       ExhibitEvent[]
  currentState ExhibitCurrentState?
  custodyState CustodyCurrentState?
  objections   ObjectionCurrentState[]

  @@unique([caseId, exhibitLabel])
  @@index([caseId])
  @@map("exhibits")
}

/// The single append-only ledger table. Rows are NEVER updated or deleted
/// after creation — enforced at the application layer: recordEvent() (Plan 2)
/// is the only writer. `sequenceNo` guarantees deterministic per-exhibit
/// replay order independent of clock skew.
model ExhibitEvent {
  id          String    @id @default(uuid())
  exhibitId   String    @map("exhibit_id")
  caseId      String    @map("case_id")
  eventType   EventType @map("event_type")
  payload     Json
  actorUserId String    @map("actor_user_id")
  sequenceNo  Int       @map("sequence_no")
  recordedAt  DateTime  @default(now()) @map("recorded_at")

  exhibit Exhibit @relation(fields: [exhibitId], references: [id])
  actor   User    @relation("EventActor", fields: [actorUserId], references: [id])

  @@unique([exhibitId, sequenceNo])
  @@index([exhibitId, recordedAt])
  @@index([caseId, eventType, recordedAt])
  @@map("exhibit_events")
}

/// One row per exhibit once it has at least one STATUS_CHANGE event.
/// Fully rebuildable by replaying exhibit_events WHERE event_type =
/// STATUS_CHANGE in sequence_no order (see Plan 7, rebuildProjections).
model ExhibitCurrentState {
  exhibitId         String        @id @map("exhibit_id")
  currentStatus     ExhibitStatus @map("current_status")
  lastStatusEventId String        @map("last_status_event_id")
  lastStatusAt      DateTime      @map("last_status_at")

  exhibit Exhibit @relation(fields: [exhibitId], references: [id])

  @@map("exhibit_current_state")
}

/// One row per objection THREAD (not per exhibit — an exhibit may have many
/// concurrently open threads). Rebuildable by replaying OBJECTION_RAISED /
/// RULING_RECORDED events grouped by payload.objectionId.
model ObjectionCurrentState {
  objectionId    String          @id @map("objection_id")
  exhibitId      String          @map("exhibit_id")
  status         ObjectionStatus @default(UNRESOLVED)
  objectingParty OfferingParty   @map("objecting_party")
  grounds        String
  raisedEventId  String          @map("raised_event_id")
  raisedAt       DateTime        @map("raised_at")
  rulingEventId  String?         @map("ruling_event_id")
  ruledAt        DateTime?       @map("ruled_at")

  exhibit Exhibit @relation(fields: [exhibitId], references: [id])

  @@index([exhibitId, status])
  @@map("objection_current_state")
}

/// One row per exhibit once it has at least one CUSTODY_TRANSFER event.
/// An exhibit with NO row here despite being ADMITTED is exactly the
/// ADMITTED_NO_CUSTODIAN discrepancy condition evaluated by Phase 3 —
/// absence is meaningful and must never be backfilled with a placeholder row.
model CustodyCurrentState {
  exhibitId              String   @id @map("exhibit_id")
  currentCustodianUserId String   @map("current_custodian_user_id")
  since                  DateTime
  lastEventId            String   @map("last_event_id")

  exhibit   Exhibit @relation(fields: [exhibitId], references: [id])
  custodian User    @relation("CurrentCustodian", fields: [currentCustodianUserId], references: [id])

  @@map("custody_current_state")
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| sandbox→host (docker) | Compose/Dockerfile build and run commands execute host-level docker processes; malformed compose config could expose unintended ports or mount unintended host paths |
| config→secrets | `.env`/compose environment variables carry DB credentials; must never be committed in a real (non-dev-placeholder) form |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Information disclosure | `.env`, `docker-compose.yml` | mitigate | `.env` is in `.gitignore` (Task 1); only `.env.example` with a clearly-labeled dev-only placeholder password (`judicialsync_dev`) is committed; `docker-compose.yml`'s inline credentials are the same dev-only placeholder, never a production secret — production (Neon) credentials are supplied via Vercel environment variables, never this file |
| T-01-02 | Tampering | `docker-compose.yml` port publishing | accept | Postgres port 5432 is published to the host for local dev convenience; acceptable at demo scale (single-case, local/sandbox environment, no production deployment via this compose file per TechArch §1.4 — production is Vercel+Neon) — owner: project maintainer, residual risk is local-only |
| T-01-03 | Denial of service | `app` service build | mitigate | Dockerfile uses a pinned Node base image (`node:20-alpine`) and a single-stage deterministic build (`npm install` against committed `package-lock.json`), avoiding unpinned `latest` tags that could silently break or bloat the build |

</threat_model>
