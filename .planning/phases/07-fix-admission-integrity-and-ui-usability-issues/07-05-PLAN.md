**Existing `useDiscrepancyCount` hook (already read from the repo — reuse exactly, do not duplicate its query):**
```typescript
export function useDiscrepancyCount() {
  // ...
  const query = useQuery<CaseDiscrepancyFlag[]>({
    queryKey: ['discrepancy-count', caseId, role],
    queryFn: async () => { /* GET /api/cases/:id/discrepancies */ },
    enabled: Boolean(caseId),
    refetchInterval: 4_000,
  });
  const flags = query.data ?? [];
  const openCount = flags.filter((f) => f.status === 'OPEN').length;
  return { ...query, flags, openCount };
}
```
